set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test)
function(kart_test name)
    add_executable(test_${name} test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE kartrecomp_build Qt6::Test)
    add_test(NAME ${name} COMMAND test_${name})
endfunction()
kart_test(unit_split)
kart_test(build_runner)
